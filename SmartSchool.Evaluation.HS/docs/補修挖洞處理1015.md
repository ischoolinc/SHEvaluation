#### 方案三：建立專門的驗證方法
**位置：** 新增私有方法

```csharp
/// <summary>
/// 驗證補修成績資料完整性
/// </summary>
/// <param name="ssr">成績記錄</param>
/// <param name="elmScore">XML元素</param>
/// <returns>驗證是否通過</returns>
private bool ValidateReScoreData(SubjectScoreRec108 ssr, XElement elmScore)
{
    if (!ssr.isScScore) return true; // 非補修成績不需要驗證
    
    string originalScore = Utility.GetAttribute(elmScore, "原始成績");
    
    // 檢查原始成績是否為空
    if (string.IsNullOrWhiteSpace(originalScore))
    {
        ssr.checkPass = false;
        // 可以記錄錯誤訊息
        return false;
    }
    
    // 檢查原始成績是否為有效數字
    if (!decimal.TryParse(originalScore, out decimal parsedScore))
    {
        ssr.checkPass = false;
        // 可以記錄錯誤訊息
        return false;
    }
    
    // 檢查原始成績是否在合理範圍內（例如：0-100）
    if (parsedScore < 0 || parsedScore > 100)
    {
        ssr.checkPass = false;
        // 可以記錄錯誤訊息
        return false;
    }
    
    return true;
}
```

**使用方式：**
```csharp
if (Utility.GetAttribute(elmScore, "是否補修成績") != null)

        if (!ValidateReScoreData(ssr, elmScore))
        {

        }
    }
}
```
